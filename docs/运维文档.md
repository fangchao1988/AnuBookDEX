# 1. 查看进程
ps aux | grep "engine-aleo" | grep -v grep

# 2. 停止（-9 = 强杀；想优雅点用 pkill 不带 -9，但引擎是常驻循环，直接 -9 干净）
pkill -9 -f "engine-aleo"
如果只想杀特定 pid:kill -9 <pid>(先 ps aux | grep engine-aleo 拿 pid)

# 3. 确认端口释放
lsof -i :9000

# 不产出 engine-aleo.bin启动 main.go的方式

按命令：pkill -9 -f "go run ./cmd/engine/aleo"

ps aux | grep "cmd/engine/aleo" | grep -v grep
